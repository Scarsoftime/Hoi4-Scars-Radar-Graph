## Calculates frame data as input
# input args (0-1 temp array):  axis^0, axis^1, axis^2, axis^3, axis^4
# output arg (temp var):        radar_graph_5axis_output
calculate_radar_graph_5axis_frame = {
	for_loop_effect = {
		start = 0 end = 5 value = v
		multiply_temp_variable = { axis^v = 25 }
		round_temp_variable = axis^v
	}

	set_temp_variable = { radar_graph_5axis_output = axis^0 }

	multiply_temp_variable = { axis^1 = 32 } # 2^5 
	add_to_temp_variable = { radar_graph_5axis_output = axis^1 }

	multiply_temp_variable = { axis^2 = 1024 } # 2^10
	add_to_temp_variable = { radar_graph_5axis_output = axis^2 }

	multiply_temp_variable = { axis^3 = 32768 } # 2^15
	add_to_temp_variable = { radar_graph_5axis_output = axis^3 }

	multiply_temp_variable = { axis^4 = 1048576 } # 2^20
	add_to_temp_variable = { radar_graph_5axis_output = axis^4 }
}

## Calculates frame data as input
# input args (temp array):   axis_init^0, axis_init^1, axis_init^2, axis_init^3, axis_init^4
#                            axis_final^0, axis_final^1, axis_final^2, axis_final^3, axis_final^4
# output args (temp array):  sector_frame^0, sector_frame^1, sector_frame^2, sector_frame^3, sector_frame^4
calculate_radar_graph_5axis_anim_frame = {
	for_each_loop = {
		array = axis_init
		multiply_temp_variable = { axis_init^i = 25 }
		round_temp_variable = axis_init^i
	}
	for_each_loop = {
		array = axis_final
		multiply_temp_variable = { axis_final^i = 25 }
		round_temp_variable = axis_final^i
	}
	
	for_loop_effect = {
		# 1-5
		start = 1 end = 6 value = v

		add_to_temp_array = { sector_frame = v }

		set_temp_variable = { temp_second_axis = v }
		modulo_temp_variable = { temp_second_axis = 5 }
		set_temp_variable = { temp_first_axis = v }
		subtract_from_temp_variable = { temp_first_axis = 1 }

		# Initial position
		set_temp_variable = { sector_scale_0 = axis_init^temp_first_axis }
		multiply_temp_variable = { sector_scale_0 = 8 } # 2^3
		set_temp_variable = { sector_scale_1 = axis_init^temp_second_axis }
		multiply_temp_variable = { sector_scale_1 = 256 } # 2^8

		# Final position
		set_temp_variable = { sector_scale_2 = axis_final^temp_first_axis }
		multiply_temp_variable = { sector_scale_2 = 8192 } # 2^13
		set_temp_variable = { sector_scale_3 = axis_final^temp_second_axis }
		multiply_temp_variable = { sector_scale_3 = 262144 } # 2^18

		set_temp_variable = { current_index = v }
		subtract_from_temp_variable = { current_index = 1 }

		add_to_temp_variable = { sector_frame^current_index = sector_scale_0 }
		add_to_temp_variable = { sector_frame^current_index = sector_scale_1 }
		add_to_temp_variable = { sector_frame^current_index = sector_scale_2 }
		add_to_temp_variable = { sector_frame^current_index = sector_scale_3 }
	}

	if = {
		limit = {
			has_country_flag = radar_graph_5axis_anim_toggle
		}
		clr_country_flag = radar_graph_5axis_anim_toggle
	}
	else = {
		set_country_flag = radar_graph_5axis_anim_toggle
	}
}

## 5-axis
maximize_radar_graph_5axis = {
	for_loop_effect = {
		start = 0 end = 5 value = v
		add_to_temp_array = { axis = 1 }
	}
	calculate_radar_graph_5axis_frame = yes
	set_variable = { radar_graph_5axis_frame = radar_graph_5axis_output }
}

randomize_radar_graph_5axis = {
	for_loop_effect = {
		start = 0 end = 5 value = v
		add_to_temp_array = { axis = 1 }
		set_temp_variable_to_random = { var = axis^v max = 1 }
	}
	calculate_radar_graph_5axis_frame = yes
	set_variable = { radar_graph_5axis_frame = radar_graph_5axis_output }
}

## 5-axis animated
maximize_radar_graph_5axis_anim = {
	## Set the initial positions with the original values
	for_loop_effect = {
		start = 0 end = 5 value = v
		add_to_temp_array = { axis_init = axis_current^v }
	}

	## Set the new values
	clear_array = axis_current
	resize_array = { array = axis_current size = 5 value = 1 }
	for_loop_effect = {
		start = 0 end = 5 value = v
		add_to_temp_array = { axis_final = axis_current^v }
	}

	## Call the scripted effect
	calculate_radar_graph_5axis_anim_frame = yes

	## Set the frame data to the output of the scripted effect
	clear_array = radar_graph_5axis_anim_frame
	for_loop_effect = {
		start = 0 end = 5 value = v
		add_to_array = { radar_graph_5axis_anim_frame = sector_frame^v }
	}
}

randomize_radar_graph_5axis_anim = {
	## Set the initial positions with the original values
	for_loop_effect = {
		start = 0 end = 5 value = v
		add_to_temp_array = { axis_init = axis_current^v }
	}

	## Set the new values
	clear_array = axis_current
	resize_array = { axis_current = 5 }
	for_loop_effect = {
		start = 0 end = 5 value = v
		set_variable_to_random = { var = axis_current^v max = 1 }
	}
	for_loop_effect = {
		start = 0 end = 5 value = v
		add_to_temp_array = { axis_final = axis_current^v }
	}

	## Call the scripted effect
	calculate_radar_graph_5axis_anim_frame = yes

	## Set the frame data to the output of the scripted effect
	clear_array = radar_graph_5axis_anim_frame
	for_loop_effect = {
		start = 0 end = 5 value = v
		add_to_array = { radar_graph_5axis_anim_frame = sector_frame^v }
	}
}
